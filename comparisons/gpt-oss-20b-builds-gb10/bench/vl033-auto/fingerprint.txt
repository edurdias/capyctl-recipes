837:def mxfp4_round_up_hidden_size_and_intermediate_size(
838:    backend: Mxfp4MoeBackend,
839:    hidden_size: int,
840:    intermediate_size: int,
841:    activation: MoEActivation | None = None,
842:) -> tuple[int, int]:
843:    """Round up hidden_size and intermediate_size based on backend requirements."""
844:    if backend in FLASHINFER_MOE_EP_MXFP4_BACKENDS:
845:        # The megakernel takes the model's dimensions and validates them itself.
846:        return hidden_size, intermediate_size
847:    if backend in B12X_BACKENDS:
848:        # b12x plans for the exact model dimensions. B12xExperts validates the
849:        # required MXFP4 block alignment before selecting the backend.
850:        return hidden_size, intermediate_size
851:    if backend == Mxfp4MoeBackend.EMULATION:
852:        # Emulation has no kernel tile; it only needs OCP MX block alignment so the
853:        # per-block scale buffers (`dim // OCP_MX_BLOCK_SIZE`) aren't floor-truncated
854:        # by a non-block-aligned TP/DP shard (e.g. 2880 // 4 = 720).
855:        intermediate_size = round_up(intermediate_size, OCP_MX_BLOCK_SIZE)
856:        hidden_size = round_up(hidden_size, OCP_MX_BLOCK_SIZE)
857:    elif backend == Mxfp4MoeBackend.DEEPGEMM_MXFP4:
858:        # DeepGEMM requires M/N/K alignment
859:        intermediate_size = round_up(intermediate_size, 128)
860:        hidden_size = round_up(hidden_size, 128)
861:    elif backend == Mxfp4MoeBackend.CUTLASS_MXFP4_MXFP4:
862:        # CUTLASS scale tiles cover 128 rows by 128 elements
863:        intermediate_size = round_up(intermediate_size, 128)
864:        hidden_size = round_up(hidden_size, 128)
865:    elif backend in (Mxfp4MoeBackend.MARLIN, Mxfp4MoeBackend.BATCHED_MARLIN):
866:        intermediate_size = round_up(intermediate_size, 128)
867:        if current_platform.is_xpu():
868:            hidden_size = round_up(hidden_size, 128)
873:        hidden_size = round_up(hidden_size, 256)
874:    elif backend in (
875:        Mxfp4MoeBackend.FLASHINFER_CUTLASS_MXFP4_BF16,
876:        Mxfp4MoeBackend.FLASHINFER_CUTLASS_MXFP4_MXFP8,
877:    ):
878:        intermediate_size = round_up(intermediate_size, 128)
879:        hidden_size = round_up(hidden_size, 128)
890:        # rather than the generic ROCm 256 round-up, which would inflate
891:        # weights and OOM.
892:        aiter_uses_128 = backend == Mxfp4MoeBackend.AITER_MXFP4_BF16
893:
894:        # matmul_ogs uses block_k=128 for MXFP4 on pre-CDNA4 GPUs.
895:        # CDNA4's F16xMXFP4 configuration uses block_k=256.
896:        triton_uses_128 = (
897:            backend == Mxfp4MoeBackend.TRITON_UNFUSED and get_cdna_version() != 4
898:        )
